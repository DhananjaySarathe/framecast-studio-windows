cmake_minimum_required(VERSION 3.24)
project(FramecastWindows VERSION 0.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(FFMPEG QUIET IMPORTED_TARGET libavformat libavcodec libavutil libswscale libswresample)
endif()
if(TARGET PkgConfig::FFMPEG)
  set(FFMPEG_LIBRARIES PkgConfig::FFMPEG)
else()
  find_path(FFMPEG_INCLUDE_DIR libavformat/avformat.h REQUIRED)
  foreach(component avformat avcodec avutil swscale swresample)
    find_library(FFMPEG_${component} NAMES ${component} REQUIRED)
    list(APPEND FFMPEG_LIBRARIES ${FFMPEG_${component}})
  endforeach()
endif()
add_library(framecast-core src/core.cpp src/output.cpp src/engine.cpp)
target_include_directories(framecast-core PUBLIC include ${FFMPEG_INCLUDE_DIR})
target_link_libraries(framecast-core PUBLIC Threads::Threads ${FFMPEG_LIBRARIES})
if(WIN32)
  target_sources(framecast-core PRIVATE src/windows_capture.cpp)
  target_compile_definitions(framecast-core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
  target_link_libraries(framecast-core PUBLIC d3d11 dxgi dwmapi windowsapp mf mfplat mfreadwrite mfuuid ole32 uuid avrt winmm)
endif()
if(MSVC)
  target_compile_options(framecast-core PUBLIC /utf-8 PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(framecast-core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(framecast-engine src/main.cpp)
target_link_libraries(framecast-engine PRIVATE framecast-core)
option(FRAMECAST_BUILD_UI "Build the Qt desktop studio" ${WIN32})
if(FRAMECAST_BUILD_UI)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Test)
  add_library(framecast-ui ui/studio_model.cpp ui/studio_window.cpp)
  target_link_libraries(framecast-ui PUBLIC framecast-core Qt6::Widgets)
  target_include_directories(framecast-ui PUBLIC ui)
  if(MSVC)
    target_compile_options(framecast-ui PRIVATE /W4 /permissive- /EHsc)
  else()
    target_compile_options(framecast-ui PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  qt_add_executable(FramecastStudio WIN32 ui/main.cpp)
  if(WIN32)
    configure_file(assets/app.rc.in ${CMAKE_CURRENT_BINARY_DIR}/app.rc @ONLY)
    target_sources(FramecastStudio PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/app.rc)
  endif()
  target_link_libraries(FramecastStudio PRIVATE framecast-ui)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(framecast-core-tests tests/core_tests.cpp)
  target_link_libraries(framecast-core-tests PRIVATE framecast-core)
  add_test(NAME core COMMAND framecast-core-tests)
  add_executable(framecast-compose-perf-tests tests/compose_perf_tests.cpp)
  target_link_libraries(framecast-compose-perf-tests PRIVATE framecast-core)
  add_test(NAME compose-1080p COMMAND framecast-compose-perf-tests)
  set_tests_properties(compose-1080p PROPERTIES TIMEOUT 120)
  add_executable(framecast-realtime-tests tests/realtime_tests.cpp)
  target_link_libraries(framecast-realtime-tests PRIVATE framecast-core)
  add_test(NAME realtime-1080p COMMAND framecast-realtime-tests)
  set_tests_properties(realtime-1080p PROPERTIES TIMEOUT 60)
  add_executable(framecast-source-failure-tests tests/source_failure_tests.cpp)
  target_link_libraries(framecast-source-failure-tests PRIVATE framecast-core)
  add_test(NAME source-failures COMMAND framecast-source-failure-tests)
  set_tests_properties(source-failures PROPERTIES TIMEOUT 60)
  add_executable(framecast-session-tests tests/session_tests.cpp)
  target_link_libraries(framecast-session-tests PRIVATE framecast-core)
  add_test(NAME live-session COMMAND framecast-session-tests)
  if(UNIX)
    add_executable(framecast-output-startup-tests tests/output_startup_tests.cpp)
    target_link_libraries(framecast-output-startup-tests PRIVATE framecast-core)
    add_test(NAME recording-backlog COMMAND framecast-output-startup-tests)
    set_tests_properties(recording-backlog PROPERTIES TIMEOUT 30)
  endif()
  if(FRAMECAST_BUILD_UI)
    add_executable(framecast-ui-tests tests/ui_tests.cpp)
    target_link_libraries(framecast-ui-tests PRIVATE framecast-ui Qt6::Test)
    add_test(NAME studio-ui COMMAND framecast-ui-tests)
    set_tests_properties(studio-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 40)
    if(WIN32)
      add_executable(framecast-capture-tests tests/windows_capture_tests.cpp)
      target_link_libraries(framecast-capture-tests PRIVATE framecast-core Qt6::Widgets)
      add_test(NAME native-window-capture COMMAND framecast-capture-tests)
      set_tests_properties(native-window-capture PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
    endif()
  endif()
endif()
